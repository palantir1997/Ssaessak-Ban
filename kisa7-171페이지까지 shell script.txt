# ── 결과 저장용 ──────────────────────────────────────────
results = []
PASS = "✅ 양호"
FAIL = "❌ 취약"
INFO = "ℹ️  정보"

def run(cmd):
    """명령어 실행 후 stdout 반환 (오류 시 빈 문자열)"""
    try:
        r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=10)
        return r.stdout.strip()
    except Exception:
        return ""

def check(code, title, status, detail):
    results.append({
        "code": code,
        "title": title,
        "status": status,
        "detail": detail
    })
    icon = status.split()[0]
    print(f"  [{code}] {icon} {title}")
    if status == FAIL:
        print(f"        → {detail}")

# U-07: 관리자 그룹(wheel/sudo)에 최소한의 계정
def check_u07():
    members = []
    for gname in ["wheel", "sudo", "admin"]:
        try:
            g = grp.getgrnam(gname)
            members.extend(g.gr_mem)
        except KeyError:
            pass
    members = list(set(members))
    if len(members) <= 2:
        check("U-07", "관리자 그룹 최소 계정", PASS, f"관리자 그룹 계정: {members}")
    else:
        check("U-07", "관리자 그룹 최소 계정", FAIL,
              f"관리자 그룹에 계정 {len(members)}개 존재: {members} → 최소화 필요")

check_u07()

# U-08: 계정이 존재하지 않는 GID 금지
def check_u08():
    bad = []
    with open("/etc/passwd") as f:
        for line in f:
            parts = line.strip().split(":")
            if len(parts) < 4:
                continue
            gid = parts[3]
            try:
                grp.getgrgid(int(gid))
            except KeyError:
                bad.append(f"{parts[0]}(GID:{gid})")
    if not bad:
        check("U-08", "존재하지 않는 GID 금지", PASS, "모든 계정의 GID가 /etc/group에 존재")
    else:
        check("U-08", "존재하지 않는 GID 금지", FAIL, f"유효하지 않은 GID 계정: {bad}")

check_u08()

# U-09: 동일한 UID 금지
def check_u09():
    uid_map = {}
    with open("/etc/passwd") as f:
        for line in f:
            parts = line.strip().split(":")
            if len(parts) < 3:
                continue
            uid = parts[2]
            name = parts[0]
            uid_map.setdefault(uid, []).append(name)
    dups = {uid: names for uid, names in uid_map.items() if len(names) > 1 and uid != "0"}
    if not dups:
        check("U-09", "동일한 UID 금지", PASS, "중복 UID 없음")
    else:
        check("U-09", "동일한 UID 금지", FAIL, f"중복 UID 발견: {dups}")

check_u09()

# U-10: 사용자 Shell 점검 (로그인 불필요 계정에 /bin/sh 등 할당 여부)
def check_u10():
    valid_shells = set(run("cat /etc/shells").splitlines())
    system_accounts = []
    with open("/etc/passwd") as f:
        for line in f:
            parts = line.strip().split(":")
            if len(parts) < 7:
                continue
            name, uid, shell = parts[0], int(parts[2]), parts[6]
            if uid < 1000 and uid != 0 and shell in valid_shells and shell not in ["/sbin/nologin", "/bin/false", "/usr/sbin/nologin"]:
                system_accounts.append(f"{name}({shell})")
    if not system_accounts:
        check("U-10", "사용자 Shell 점검", PASS, "시스템 계정에 불필요한 Shell 없음")
    else:
        check("U-10", "사용자 Shell 점검", FAIL,
              f"로그인 Shell 할당된 시스템 계정: {system_accounts}")

check_u10()

# U-11: 세션 종료 시간 설정 (TMOUT)
def check_u11():
    tmout = run("grep -r 'TMOUT' /etc/profile /etc/profile.d/ /etc/bash.bashrc 2>/dev/null")
    if tmout:
        # TMOUT 값 추출
        val = run("grep -r 'TMOUT' /etc/profile /etc/profile.d/ /etc/bash.bashrc 2>/dev/null | grep -oP 'TMOUT=\\K[0-9]+'")
        if val and int(val.split()[0]) <= 600:
            check("U-11", "세션 종료 시간 설정", PASS, f"TMOUT={val.split()[0]}초 설정됨")
        else:
            check("U-11", "세션 종료 시간 설정", FAIL, f"TMOUT 값이 너무 크거나 미설정: {val}")
    else:
        check("U-11", "세션 종료 시간 설정", FAIL, "TMOUT 미설정 → /etc/profile에 'export TMOUT=600' 추가 필요")

check_u11()

# U-12: 패스워드 암호화 알고리즘 (SHA-512 권고)
def check_u12():
    algo = run("grep -E '^ENCRYPT_METHOD|^SHA_CRYPT' /etc/login.defs")
    shadow_sample = run("awk -F: '$2 ~ /^\\$/ {print $2}' /etc/shadow 2>/dev/null | head -1")
    if "$6$" in (shadow_sample or ""):
        check("U-12", "패스워드 암호화 알고리즘", PASS, "SHA-512($6$) 사용 중")
    elif "$5$" in (shadow_sample or ""):
        check("U-12", "패스워드 암호화 알고리즘", PASS, "SHA-256($5$) 사용 중")
    elif shadow_sample:
        check("U-12", "패스워드 암호화 알고리즘", FAIL, f"취약한 암호화 알고리즘 사용: {shadow_sample[:10]}...")
    else:
        check("U-12", "패스워드 암호화 알고리즘", INFO, f"shadow 확인 불가 (권한 부족 또는 설정: {algo})")

check_u12()

def check_file_perm(code, title, path, max_perm, expected_owner="root"):
    """파일 소유자 및 권한 통합 점검"""
    if not os.path.exists(path):
        check(code, title, INFO, f"{path} 파일 없음")
        return
    s = os.stat(path)
    perm = oct(s.st_mode)[-3:]
    owner = pwd.getpwuid(s.st_uid).pw_name
    issues = []
    if owner != expected_owner:
        issues.append(f"소유자={owner}(기대값:{expected_owner})")
    if int(perm, 8) > int(max_perm, 8):
        issues.append(f"권한={perm}(최대허용:{max_perm})")
    if issues:
        check(code, title, FAIL, f"{path}: {', '.join(issues)}")
    else:
        check(code, title, PASS, f"{path}: 소유자={owner}, 권한={perm}")

# U-22: SUID/SGID 파일 점검
def check_u22():
    out = run("find / -xdev \\( -perm -4000 -o -perm -2000 \\) -type f 2>/dev/null")
    files = [f for f in out.splitlines() if f]
    # 기본 허용 목록
    allowed = {"/usr/bin/sudo", "/usr/bin/passwd", "/usr/bin/su",
               "/usr/bin/newgrp", "/usr/bin/chsh", "/usr/bin/chfn",
               "/bin/ping", "/usr/bin/ping", "/usr/bin/pkexec"}
    unknown = [f for f in files if f not in allowed]
    if not unknown:
        check("U-22", "SUID/SGID 설정 파일 점검", PASS, f"허용된 SUID/SGID 파일만 존재 ({len(files)}개)")
    else:
        check("U-22", "SUID/SGID 설정 파일 점검", FAIL,
              f"비허용 SUID/SGID 파일 {len(unknown)}개: {unknown[:5]}{'...' if len(unknown)>5 else ''}")

check_u22()

# U-30: 홈 디렉토리 소유자 및 권한
def check_u30():
    issues = []
    with open("/etc/passwd") as f:
        for line in f:
            parts = line.strip().split(":")
            if len(parts) < 7:
                continue
            name, uid, homedir = parts[0], int(parts[2]), parts[5]
            if uid < 1000 or not homedir or not os.path.exists(homedir):
                continue
            s = os.stat(homedir)
            owner_uid = s.st_uid
            perm = oct(s.st_mode)[-3:]
            if owner_uid != int(parts[2]):
                issues.append(f"{name}: 홈디렉토리 소유자 불일치")
            if int(perm, 8) & 0o022:
                issues.append(f"{name}: 홈디렉토리 권한 {perm} (타인 쓰기 가능)")
    if not issues:
        check("U-30", "홈 디렉토리 소유자 및 권한", PASS, "홈 디렉토리 권한 적절")
    else:
        check("U-30", "홈 디렉토리 소유자 및 권한", FAIL, "; ".join(issues[:3]))

check_u30()

def check_service_disabled(code, title, service_names):
    """서비스 비활성화 여부 점검"""
    running = []
    for svc in service_names:
        status = run(f"systemctl is-active {svc} 2>/dev/null")
        if status == "active":
            running.append(svc)
        # xinetd 방식도 확인
        inetd = run(f"grep -l '{svc}' /etc/xinetd.d/ 2>/dev/null")
        if inetd:
            disabled = run(f"grep 'disable.*yes' /etc/xinetd.d/{svc} 2>/dev/null")
            if not disabled:
                running.append(f"{svc}(xinetd)")
    if not running:
        check(code, title, PASS, f"{service_names} 서비스 비활성화 확인")
    else:
        check(code, title, FAIL, f"활성화된 서비스: {running} → 비활성화 필요")

# U-66: 로그 디렉토리 소유자 및 권한
def check_u66():
    log_dirs = ["/var/log"]
    issues = []
    for path in log_dirs:
        if not os.path.exists(path):
            continue
        s = os.stat(path)
        perm = oct(s.st_mode)[-3:]
        owner = pwd.getpwuid(s.st_uid).pw_name
        if owner != "root":
            issues.append(f"{path}: 소유자={owner}")
        if int(perm, 8) & 0o002:
            issues.append(f"{path}: other 쓰기 가능({perm})")
    if not issues:
        check("U-66", "로그 디렉토리 소유자 및 권한", PASS, "/var/log 권한 적절")
    else:
        check("U-66", "로그 디렉토리 소유자 및 권한", FAIL, "; ".join(issues))

check_u66()

# Apache/웹서버 실행 계정 점검
def check_web_user():
    apache_user = run("grep -E '^User|^www-data' /etc/apache2/apache2.conf /etc/httpd/conf/httpd.conf 2>/dev/null | head -1")
    nginx_user = run("grep '^user' /etc/nginx/nginx.conf 2>/dev/null | head -1")
    conf = apache_user or nginx_user
    if "root" in (conf or ""):
        check("WEB-01", "웹서버 실행 계정", FAIL, "웹서버가 root로 실행 중 → 전용 계정으로 변경 필요")
    else:
        check("WEB-01", "웹서버 실행 계정", PASS, f"웹서버 계정: {conf or 'www-data/apache(기본)'}")

check_web_user()

# MySQL/MariaDB root 패스워드 설정 여부
def check_db_root():
    result = run("mysql -u root --connect-timeout=3 -e 'SELECT 1' 2>&1")
    if "Access denied" in result:
        check("DB-01", "DB root 패스워드 설정", PASS, "MySQL root 패스워드 설정됨")
    elif "ERROR" not in result and result:
        check("DB-01", "DB root 패스워드 설정", FAIL, "MySQL root 패스워드 없이 접속 가능!")
    else:
        check("DB-01", "DB root 패스워드 설정", INFO, "MySQL 미실행 또는 확인 불가")

check_db_root()

# SSH 설정 점검
def check_ssh():
    sshd_conf = "/etc/ssh/sshd_config"
    if not os.path.exists(sshd_conf):
        check("SSH-01", "SSH 보안 설정", INFO, "sshd_config 없음")
        return
    conf = open(sshd_conf).read()
    issues = []
    if "PermitRootLogin yes" in conf:
        issues.append("root SSH 로그인 허용됨")
    if "PasswordAuthentication yes" in conf and "PubkeyAuthentication yes" not in conf:
        issues.append("패스워드 인증만 사용 중 (키 인증 권고)")
    if "Protocol 1" in conf:
        issues.append("SSH Protocol 1 사용 중 (취약)")
    if not issues:
        check("SSH-01", "SSH 보안 설정", PASS, "SSH 보안 설정 적절")
    else:
        check("SSH-01", "SSH 보안 설정", FAIL, "; ".join(issues))

check_ssh()
